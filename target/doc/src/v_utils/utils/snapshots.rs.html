1struct PlotData {
2	scale: f64,
3	offset: f64,
4	blocks: [char; 9],
5}
6
7static SINGLE_PLOT_WIDTH: usize = 90;
8static SINGLE_PLOT_HEIGHT: usize = 12;
9
10impl PlotData {
11	fn new(min_val: f64, max_val: f64, height: usize) -> Self {
12		let data_range = max_val - min_val;
13		let plot_range = (height * 8) as f64;
14		let scale = plot_range / data_range;
15		let offset = min_val * scale;
16		let blocks = [' ', '▁', '▂', '▃', '▄', '▅', '▆', '▇', '█'];
17		//let braille = ['⠀', '⡀', '⡄', '⡆', '⢀', '⢠', '⣀', '⣆', '⣠', '⣤', '⣦', '⣴', '⣶', '⣷', '⣾', '⣿'];
18		PlotData { scale, offset, blocks }
19	}
20
21	fn get_block_index(&self, val: f64, i: usize) -> usize {
22		let scaled_val = val * self.scale - self.offset;
23		(scaled_val - i as f64 * 8.0).clamp(0.0, 8.0) as usize
24	}
25
26	/// Raise by the smallest step (▁)
27	fn raise_plot(&mut self) {
28		self.offset -= 1.0;
29	}
30}

33pub struct SnapshotP {
34	prices: Vec<f64>,
35	secondary_pane: Option<Vec<Option<f64>>>,
36	width: usize,
37	height: usize,
38}
39/// Very not DRY
40impl SnapshotP {
41	pub fn build<T: Into<f64> + Copy>(prices: &[T]) -> Self {
42		SnapshotP {
43			prices: prices.iter().map(|x| (*x).into()).collect(),
44			secondary_pane: None,
45			width: SINGLE_PLOT_WIDTH,
46			height: SINGLE_PLOT_HEIGHT,
47		}
48	}
49
50	/// Height is always 2/5 that of the main pane
51	pub fn secondary_pane_optional<T: Into<f64> + Copy>(self, secondary_pane: Vec<Option<T>>) -> Self {
52		SnapshotP {
53			secondary_pane: Some(secondary_pane.iter().map(|x| x.map(|x| x.into())).collect()),
54			..self
55		}
56	}
57
58	/// Height is always 2/5 that of the main pane
59	pub fn secondary_pane<T: Into<f64> + Copy>(self, secondary_pane: Vec<T>) -> Self {
60		SnapshotP {
61			secondary_pane: Some(secondary_pane.iter().map(|x| Some((*x).into())).collect()),
62			..self
63		}
64	}
65
66	/// Default width is `90`
67	pub fn width(self, width: usize) -> Self {
68		SnapshotP { width, ..self }
69	}
70
71	/// Set height of the main pane. Secondary pane's height is automatically determined. Default height is `20`
72	pub fn height_main_pane(self, height: usize) -> Self {
73		SnapshotP { height, ..self }
74	}
75
76	/// # Panics
77	/// Meant to be used only in tests, so if any input params are incorrect we panic.
78	pub fn draw(self) -> String {
79		let main_section = Self::plot_p(self.prices, self.width, self.height); // main must be plot_p, because first and last on it can never be empty.
80		let mut out = main_section;
81		if let Some(secondary_pane) = self.secondary_pane {
82			let separator = "─".repeat(self.width);
83			let secondary_section = Self::plot_p_optional(secondary_pane, self.width, (self.height * 3) / 5);
84			out.push_str(&format!("\n{separator}\n{secondary_section}"));
85		}
86		out
87	}

89	fn plot_p_optional(prices: Vec<Option<f64>>, width: usize, height: usize) -> String {
90		if prices.is_empty() {
91			return " ".repeat(width).repeat(height);
92		}
93		let non_empty_prices = prices.iter().filter_map(|x| *x).collect::<Vec<f64>>();
94
95		let min_val = non_empty_prices.iter().fold(f64::INFINITY, |a, &b| a.min(b));
96		let max_val = non_empty_prices.iter().fold(f64::NEG_INFINITY, |a, &b| a.max(b));
97
98		let min_step = (max_val - min_val) / 100.0;
99		let f_len = min_step.to_string().split('.').collect::<Vec<&str>>()[1].chars().take_while(|&c| c == '0').count() + 1;
100		let max_str = format!("{:.f_len$}", max_val).trim_end_matches(".0").to_string();
101		let min_str = format!("{:.f_len$}", min_val).trim_end_matches(".0").to_string();
102		let side_panel_width = max_str.len().max(min_str.len());
103		let mut side_panel = String::with_capacity(height * side_panel_width);
104		for i in 0..height {
105			if i == 0 {
106				side_panel.push_str(&format!("{max_str}\n"));
107			} else if i == height - 1 {
108				side_panel.push_str(&format!("{min_str}\n"));
109			} else {
110				side_panel.push_str(&format!("{:>side_panel_width$}\n", " "));
111			}
112		}
113		side_panel.pop(); // remove last newline
114
115		if (max_val - min_val).abs() < f64::EPSILON {
116			return " ".repeat(width).repeat(height);
117		}
118
119		let mut plot_data = PlotData::new(min_val, max_val, height);
120		plot_data.raise_plot(); // here we always want to reise to be able to distinguish between empty and non-empty prices
121
122		let mut plot = Vec::with_capacity(height);
123		for i in (0..height).rev() {
124			let row: String = (0..width)
125				.map(|j| {
126					let index = (j as f64 * prices.len() as f64 / width as f64) as usize;
127					match prices[index] {
128						Some(val) => {
129							let block_index = plot_data.get_block_index(val, i);
130							plot_data.blocks[block_index]
131						}
132						None => ' ',
133					}
134				})
135				.collect();
136			plot.push(row);
137		}
138
139		join_str_blocks_v(plot.join("\n"), side_panel)
140	}
141
142	fn plot_p(prices: Vec<f64>, width: usize, height: usize) -> String {
143		if prices.is_empty() {
144			panic!("prices are empty");
145		}
146
147		let min_val = prices.iter().fold(f64::INFINITY, |a, &b| a.min(b));
148		let max_val = prices.iter().fold(f64::NEG_INFINITY, |a, &b| a.max(b));
149		if (max_val - min_val).abs() < f64::EPSILON {
150			return " ".repeat(width).repeat(height);
151		}
152
153		let min_step = (max_val - min_val) / 100.0;
154		let f_len = min_step.to_string().split('.').collect::<Vec<&str>>()[1].chars().take_while(|&c| c == '0').count() + 1;
155		let max_str = format!("{:.f_len$}", max_val).trim_end_matches(".0").to_string();
156		let min_str = format!("{:.f_len$}", min_val).trim_end_matches(".0").to_string();
157		let side_panel_width = max_str.len().max(min_str.len());
158		let mut side_panel = String::with_capacity(height * side_panel_width);
159		for i in 0..height {
160			if i == 0 {
161				side_panel.push_str(&format!("{max_str}\n"));
162			} else if i == height - 1 {
163				side_panel.push_str(&format!("{min_str}\n"));
164			} else {
165				side_panel.push_str(&format!("{:>side_panel_width$}\n", " "));
166			}
167		}
168		side_panel.pop(); // remove last newline
169
170		let mut plot_data = PlotData::new(min_val, max_val, height);
171
172		// Check if we need to raise the plot
173		let first_block = plot_data.get_block_index(prices[0], height - 1);
174		let last_block = plot_data.get_block_index(prices[prices.len() - 1], height - 1);
175		if first_block == 0 || last_block == 0 {
176			plot_data.raise_plot();
177		}
178
179		let mut plot = Vec::with_capacity(height);
180
181		for i in (0..height).rev() {
182			let row: String = (0..width)
183				.map(|j| {
184					let index = (j as f64 * prices.len() as f64 / width as f64) as usize;
185					let val = prices[index];
186					let block_index = plot_data.get_block_index(val, i);
187					plot_data.blocks[block_index]
188				})
189				.collect();
190			plot.push(row);
191		}
192
193		join_str_blocks_v(plot.join("\n"), side_panel)
194	}
195}

197fn join_str_blocks_v(left: String, right: String) -> String {
198	assert_eq!(left.split('\n').count(), right.split('\n').count());
199	left.lines().zip(right.lines()).map(|(l, r)| format!("{}{}", l, r)).collect::<Vec<String>>().join("\n")
200}

202/// # Panics
203/// if ordinals on orders are outside of prices or not ascending.
204///
205/// # Blocker
206/// Until better fonts, distinctions between price formats, multiple order lines at a time & order types, actual timeframes, are all extremely problematic; so their implementation is postponed.
207///
208/// # Architecture
209/// Uses [SnapshotP] to build the plot, for finer control use it instead.
210pub fn snapshot_plot_orders<T: Into<f64> + Copy>(prices: &[T], orders: &[(usize, Option<T>)]) -> String {
211	let prices = prices.iter().map(|x| (*x).into()).collect::<Vec<f64>>();
212	let orders = orders.iter().map(|(i, x)| (*i, x.map(|x| x.into()))).collect::<Vec<(usize, Option<f64>)>>();
213	assert!(orders.iter().all(|(i, _)| *i < prices.len()));
214	assert!(orders.windows(2).all(|w| w[0].0 < w[1].0));
215
216	let mut order_points = Vec::with_capacity(prices.len());
217	let mut last_order: (usize, Option<f64>) = (0, None);
218	for (i, order) in orders.iter() {
219		order_points.extend((last_order.0..*i).map(|_| last_order.1));
220		last_order = (*i, *order);
221	}
222	order_points.extend((last_order.0..prices.len()).map(|_| last_order.1));
223
224	SnapshotP::build(&prices).secondary_pane_optional(order_points).draw()
225}
