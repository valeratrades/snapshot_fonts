4/// Doesn't support negative timestamps
5pub fn guess_timestamp_unsafe(timestamp: String) -> Result<Timestamp> {
6	// Try parsing as ISO 8601 format
7	if let Ok(dt) = timestamp.parse::<Timestamp>() {
8		return Ok(dt);
9	}
10
11	// Try guessing the denominator
12	if let Ok(num) = timestamp.parse::<u64>() {
13		let len = timestamp.len();
14		let nanos = match len {
15			10 => num * 1_000_000_000,
16			13 => num * 1_000_000,
17			16 => num * 1_000,
18			19 => num,
19			_ => bail!("Invalid timestamp length for guessing: {len}\nTimestamp: {timestamp}"),
20		};
21		return Ok(Timestamp::from_nanosecond(nanos as i128).unwrap());
22	}
23
24	bail!("Couldn't parse timestamp: {}", timestamp)
25}
